                    fa.area_id,
                    fa.farmer_reg_number,
                    fld.loan_item_id,
                    fld.quantity
                FROM    farmer_loan_requests flr
                            JOIN    farmer_loan_main flm
                                    on flr.id = flm.farmer_loan_request_id
                                        AND flr.approval_status = 'APPROVED'
                            JOIN    farmer_loan_details fld
                                    on flm.id = fld.loan_main_id
                            JOIN    farmer_agreement fa
                                    ON flr.farmer_agreement_id = fa.id
                                        AND fa.season_id = :seasonId
                                        AND (:areaId = 0 or fa.area_id = :areaId)
            )

                        l.*

        ) as d

    select
        fa.area_id,

) as v

                         ON d.loan_item_id = bd.loan_item_id
