        Mod(sum(case when qqc.category_id=1 and qqo.question_option_id=4 and qq.question_id=3 then cast(qqa.answer as numeric) end) , 12) seedLateLobInch,

        sum(case when qqc.category_id=4 and qqom1.option_id=7 and qqo.question_option_id=11 and qq.question_id=8 then cast(qqa.answer as numeric) end) pakkaNew161616
from    (select * from farmer_agreement
         where season_id=:seasonId
             and :areaId=0 or area_id=:areaId
             and :zoneId=0 or zone_id=:zoneId
             and :subAreaId=0 or sub_Area_Id=:subAreaId
             and :villageId=0 or village_id=:villageId
         ) fa
            join    qi_question_answer qqa on fa.id=qqa.farmer_agreement_id
            join    qi_question_option_map qqom
                    on qqa.question_option_map_id = qqom.question_option_map_id
                        and     qqa.question_id = qqom.question_id
            join    qi_question_option qqo on qqom.option_id = qqo.question_option_id
            join    qi_question qq on qqa.question_id = qq.question_id
            join    qi_question_category qqc on qq.question_category_id = qqc.category_id
            and     (case when :categoryId is not null then qqc.category_id = :categoryId else true end)
            left    join qi_question_option_map qqom1 on qqom.parent_id=qqom1.question_option_map_id
group   by grouping sets ((area_id, area_name, zone_id, zone_name, sub_area_id, sub_area_name, village_id, village_name,
                           farmer_reg_number, farmer_name, farmer_father_name, mobile_no, land_acreage),
                          (area_id, area_name, zone_id, zone_name, sub_area_id, sub_area_name, village_id,village_name),
                          (area_id, area_name, zone_id, zone_name, sub_area_id, sub_area_name),
                          (area_id, area_name, zone_id, zone_name),
                          (area_id, area_name),
                          ()
                         )
order   by area_id, zone_id, sub_area_id, village_id
